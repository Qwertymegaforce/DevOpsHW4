
#### 2.3 пароль генерируется случайно при установке и не хранится в репозитории

```console
$ cat scripts/create-db-secret.sh
#!/bin/sh
set -eu

NAMESPACE="${NAMESPACE:-counter}"

if kubectl -n "$NAMESPACE" get secret postgres-auth >/dev/null 2>&1; then
  echo "secret/postgres-auth уже существует в namespace $NAMESPACE, пропускаю"
  exit 0
fi

kubectl -n "$NAMESPACE" create secret generic postgres-auth \
  --from-literal=POSTGRES_USER=counter \
  --from-literal=POSTGRES_DB=counter \
  --from-literal=POSTGRES_PASSWORD="$(openssl rand -base64 24 | tr -d '/+=')"

$ ./scripts/create-db-secret.sh
secret/postgres-auth created

# значение пароля не печатаем — только ключи и длину:
$ kubectl -n counter get secret postgres-auth -o go-template='{{range $k, $v := .data}}{{$k}}: {{len ($v | base64decode)}} символов{{"\n"}}{{end}}'
POSTGRES_DB: 7 символов
POSTGRES_PASSWORD: 30 символов
POSTGRES_USER: 7 символов

# в репозитории пароля нет: в манифестах только ссылки на Secret
$ grep -rn -i 'password' k8s/ charts/ | grep -v '^charts/counter/templates/NOTES'
k8s/tools/pg-client.yaml:13:        - name: PGPASSWORD
k8s/tools/pg-client.yaml:17:              key: POSTGRES_PASSWORD
charts/counter/templates/postgres.yaml:25:  POSTGRES_PASSWORD: {{ index $existing.data "POSTGRES_PASSWORD" | quote }}
charts/counter/templates/postgres.yaml:27:  POSTGRES_PASSWORD: {{ randAlphaNum 24 | b64enc | quote }}

```
